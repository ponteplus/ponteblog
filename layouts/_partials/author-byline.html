Parâmetros: page (obrigatório), format: inline | rich (default inline) */ -}} {{- $page := .page -}} {{- $format := .format | default "inline" -}} {{- $slugs := $page.Params.authors -}} {{- $registry := site.Data.authors -}} {{- if $slugs -}} {{- $lazyLoading := site.Params.enableImageLazyLoading | default true -}} {{- range $i, $slug := $slugs -}} {{- if $i -}},{{- end -}} {{- $entry := index $registry $slug -}} {{- $name := cond $entry $entry.name ($slug | humanize | title) -}} {{- $url := printf "/authors/%s/" $slug | relURL -}} {{- $image := "" -}} {{- if $entry }}{{ $image = $entry.image }}{{ end -}} {{- if eq $format "rich" -}} {{- with $image }} {{- end -}} {{ $name }} {{- else -}} {{ $name }} {{- end -}} {{- end -}} {{- end -}}
